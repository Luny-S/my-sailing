% Inline underlined field: \Fld{name}{width}
\newcommand{\Fld}[2]{%
  \TextField[name=#1,width=#2,height=11pt,charsize=9pt,
    borderstyle=U,borderwidth=0.6,bordercolor={0.55 0.55 0.55},
    backgroundcolor={},color={0 0 0}]{}}
% Borderless field that fills a table cell: \CFld{name}{width}
\newcommand{\CFld}[2]{%
  \TextField[name=#1,width=#2,height=12pt,charsize=9pt,
    borderwidth=0,bordercolor={},backgroundcolor={},color={0 0 0}]{}}
% Multiline field (framed light box): \MFld{name}{width}{height}
\newcommand{\MFld}[3]{%
  \TextField[name=#1,width=#2,height=#3,charsize=9pt,multiline=true,
    borderstyle=U,borderwidth=0.6,bordercolor={0.55 0.55 0.55},
    backgroundcolor={},color={0 0 0}]{}}
% Label + stretchy field that fills the rest of the current line/cell width:
% \LF{label}{name}  — used to build invisible 50/50 label-field tables.
\newlength{\lftmp}
\newcommand{\LF}[2]{%
  \settowidth{\lftmp}{#1\,}%
  #1\,\TextField[name=#2,width=\dimexpr\linewidth-\lftmp-2pt\relax,height=11pt,
    charsize=9pt,borderstyle=U,borderwidth=0.6,bordercolor={0.55 0.55 0.55},
    backgroundcolor={},color={0 0 0}]{}}
